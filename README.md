    <tr>
        <th>Pin/signal</th>
        <th>Function</th>
        <th>Wire color convention</th>
    </tr>
    <tr>
        <td>MOSI</td>

        <td>blue</td>
    </tr>
    <tr>
        <td>MISO</td>

        <td>green</td>
    </tr>

        <td>SCK</td>
        <td>serial clock</td>
        <td>yellow</td>
    </tr>
    <tr>
        <td>CS/SS</td>
        <td>chip select</td>
        <td>purple</td>
    </tr>
</table>
